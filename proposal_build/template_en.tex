\documentclass[12pt,a4paper,oneside]{book}

% ========== Fonts (XeLaTeX) ==========
\usepackage{fontspec}
% NOTE: macOS "Times New Roman" under XeLaTeX embeds a broken ToUnicode map that
% encodes every semicolon as U+037E (Greek question mark). The PDF looks correct
% but text extraction (copy-paste, Turnitin) reads Greek homoglyphs, which trips
% Turnitin's "replaced characters" integrity flag. TeX Gyre Termes is a
% metric-compatible Times clone with a correct Unicode map and identical look.
\setmainfont{texgyretermes-regular.otf}[
  BoldFont       = texgyretermes-bold.otf,
  ItalicFont     = texgyretermes-italic.otf,
  BoldItalicFont = texgyretermes-bolditalic.otf ]
